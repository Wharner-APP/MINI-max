cmake_minimum_required(VERSION 3.21)
project(minimax VERSION 4.10.26.0 LANGUAGES C CXX)
if(WIN32)
  enable_language(RC)
endif()

set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 20)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(MINIMAX_BUILD_TESTS "Build unit tests" ON)
set(MINIMAX_PLATFORM_ID "" CACHE STRING "Update channel id: minimaxWinX64, minimaxLinuxX86 or minimaxMacX64 (auto-detected if empty)")

# ---- release metadata (shown in file properties on Windows) -------------------
set(MM_APP_NAME      "MINI max")
set(MM_COMPANY       "Wharner-APP")
set(MM_COPYRIGHT     "Wharner Group")
set(MM_WEBSITE       "https://wharner-official-app.tilda.ws/")
set(MM_GITHUB       "https://github.com/Wharner-APP/MINI-max")
set(MM_VERSION_STR   "04.10.26.0")
set(MM_VERSION_RC    "4,10,26,0")

# ---- update channel id -------------------------------------------------------
if(NOT MINIMAX_PLATFORM_ID)
  if(WIN32)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
      set(MINIMAX_PLATFORM_ID minimaxWinX64)
    else()
      message(FATAL_ERROR "MINI max currently supports Windows 64-bit only")
    endif()
  elseif(APPLE)
    set(MINIMAX_PLATFORM_ID minimaxMacX64)
  else()
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
      set(MINIMAX_PLATFORM_ID minimaxLinuxX86)
    else()
      message(FATAL_ERROR "MINI max currently supports Linux 32-bit (i386) only")
    endif()
  endif()
endif()
message(STATUS "MINI max platform id: ${MINIMAX_PLATFORM_ID}")

# ---- CaptchaAPI.cfg: read at configure time, compiled into the binary, never shipped ----
set(MM_CAPTCHA_KEY "")
set(_cfg "${PROJECT_SOURCE_DIR}/CaptchaAPI.cfg")
if(EXISTS "${_cfg}")
  file(STRINGS "${_cfg}" _lines)
  foreach(_l IN LISTS _lines)
    if(_l MATCHES "^[ \t]*API_KEY[ \t]*=[ \t]*(.*)$")
      string(STRIP "${CMAKE_MATCH_1}" MM_CAPTCHA_KEY)
    endif()
  endforeach()
  if(MM_CAPTCHA_KEY MATCHES "^PUT_YOUR")
    set(MM_CAPTCHA_KEY "")
  endif()
  string(LENGTH "${MM_CAPTCHA_KEY}" _klen)
  message(STATUS "CaptchaAPI.cfg found, key length: ${_klen}")
else()
  message(STATUS "CaptchaAPI.cfg not found - building without an embedded captcha key")
endif()

# ---- Qt 6 (preferred) or Qt 5.15 (used for the 32-bit Windows build) ---------------
set(MINIMAX_QT_MAJOR "" CACHE STRING "Force Qt major version (5 or 6); empty = prefer 6, fall back to 5")
if(MINIMAX_QT_MAJOR)
  find_package(QT NAMES Qt${MINIMAX_QT_MAJOR} REQUIRED COMPONENTS Core)
else()
  find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
endif()
set(QTV Qt${QT_VERSION_MAJOR})
find_package(${QTV} REQUIRED COMPONENTS Core Network Widgets Svg Multimedia MultimediaWidgets)
message(STATUS "Using ${QTV} ${${QTV}_VERSION}")

# ---- third party ----------------------------------------------------------------
include(FetchContent)
FetchContent_Declare(tomlplusplus GIT_REPOSITORY https://github.com/marzer/tomlplusplus.git GIT_TAG v3.4.0)
FetchContent_Declare(libsodium_cmake GIT_REPOSITORY https://github.com/robinlinden/libsodium-cmake.git GIT_TAG master)
set(SODIUM_DISABLE_TESTS ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(tomlplusplus libsodium_cmake)

configure_file(common/build_info.h.in ${CMAKE_BINARY_DIR}/generated/mm/build_info.h @ONLY)

if(MINIMAX_BUILD_TESTS)
  enable_testing()
endif()
add_subdirectory(common)
add_subdirectory(client)
add_subdirectory(updater)
